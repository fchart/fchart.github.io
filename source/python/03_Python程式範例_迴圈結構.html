print("\n題目79：找出範圍內的完全平方數")
start = int(input("請輸入起始值："))
end = int(input("請輸入結束值："))
i = 1
while i ** 2 <= end:
    square = i ** 2
    if square >= start:
        print(square, end=" ")
    i += 1
print()
題目 80：簡易 ATM 系統
print("\n題目80：簡易 ATM 系統")
balance = 10000
while True:
    print(f"\n目前餘額：{balance} 元")
    print("1.存款  2.提款  3.查詢  4.離開")
    choice = int(input("請選擇功能："))
    if choice == 1:
        amount = int(input("請輸入存款金額："))
        balance += amount
        print(f"存款成功，餘額 = {balance} 元")
    elif choice == 2:
        amount = int(input("請輸入提款金額："))
        if amount > balance:
            print("餘額不足")
        else:
            balance -= amount
            print(f"提款成功，餘額 = {balance} 元")
    elif choice == 3:
        print(f"目前餘額 = {balance} 元")
    elif choice == 4:
        print("謝謝使用，再見！")
        break
    else:
        print("無效選擇")
