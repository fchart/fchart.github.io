words = ["apple", "banana", "avocado", "cherry", "apricot"]
grouped = {}
for word in words:
    first_letter = word[0]
    if first_letter not in grouped:
        grouped[first_letter] = []
    grouped[first_letter].append(word)
print(f"分組：{grouped}")

題目 78：資料轉換管道
print("\n題目78：資料轉換管道")
text = "hello world python programming"
words = text.split()
# 步驟1：過濾長度 > 5
filtered = [w for w in words if len(w) > 5]
# 步驟2：轉大寫
uppercased = [w.upper() for w in filtered]
# 步驟3：建立字典
result = {w: len(w) for w in uppercased}
print(f"原文：{text}")
print(f"過濾：{filtered}")
print(f"大寫：{uppercased}")
print(f"字典：{result}")
題目 79：學生成績分析系統
print("\n題目79：成績分析系統")
students = [
    {"name": "Alice", "scores": [85, 90, 88]},
    {"name": "Bob", "scores": [78, 82, 80]},
    {"name": "Charlie", "scores": [92, 95, 93]}
]

# 計算平均分
averages = {s["name"]: sum(s["scores"]) / len(s["scores"]) for s in students}

# 及格名單
passed = [s["name"] for s in students if sum(s["scores"]) / len(s["scores"]) >= 80]

# 最高分
max_avg = max(averages.values())
top_student = [name for name, avg in averages.items() if avg == max_avg]

print(f"學生資料：{students}")
print(f"平均分：{averages}")
print(f"及格名單：{passed}")
print(f"最高分學生：{top_student}")
題目 80：綜合電商系統
print("\n題目80：電商訂單處理系統")
products = {
    "P001": {"name": "筆記本", "price": 25, "stock": 100},
    "P002": {"name": "原子筆", "price": 10, "stock": 200},
    "P003": {"name": "橡皮擦", "price": 5, "stock": 150}
}

orders = [
    {"product_id": "P001", "quantity": 3},
    {"product_id": "P002", "quantity": 5},
    {"product_id": "P001", "quantity": 2}
]

# 計算訂單總價
order_totals = []
for order in orders:
    new_order = dict(order)
    new_order["product_name"] = products[order["product_id"]]["name"]
    new_order["unit_price"] = products[order["product_id"]]["price"]
    new_order["subtotal"] = products[order["product_id"]]["price"] * order["quantity"]
    order_totals.append(new_order)

# 庫存更新（模擬）
stock_updates = {}
for order in orders:
    pid = order["product_id"]
    if pid in stock_updates:
        stock_updates[pid] += order["quantity"]
    else:
        stock_updates[pid] = order["quantity"]

updated_stock = {
    pid: products[pid]["stock"] - stock_updates.get(pid, 0)
    for pid in products
}

# 總營收
total_revenue = sum(item["subtotal"] for item in order_totals)

# 熱賣商品（銷量 > 3）
hot_items = [
    products[pid]["name"] 
    for pid, qty in stock_updates.items() 
    if qty > 3
]

print("=== 訂單處理 ===")
print(f"訂單明細：")
for item in order_totals:
    print(f"  {item['product_name']} x {item['quantity']} = {item['subtotal']} 元")

print(f"\n總營收：{total_revenue} 元")
print(f"庫存更新：{updated_stock}")
print(f"熱賣商品：{hot_items}")
