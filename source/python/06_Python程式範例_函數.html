def fibonacci(n):
    if n <= 0:
        return []
    elif n == 1:
        return [0]
    elif n == 2:
        return [0, 1]
    
    fib = [0, 1]
    for i in range(2, n):
        fib.append(fib[i-1] + fib[i-2])
    return fib

def bubble_sort(numbers):
    sorted_list = numbers.copy()
    n = len(sorted_list)
    for i in range(n):
        for j in range(0, n - i - 1):
            if sorted_list[j] > sorted_list[j + 1]:
                temp = sorted_list[j]
                sorted_list[j] = sorted_list[j + 1]
                sorted_list[j + 1] = temp
    return sorted_list

data = [64, 34, 25, 12, 22, 11, 90]
print(f"排序前：{data}")
print(f"排序後：{bubble_sort(data)}")

def average(numbers):
    return sum(numbers) / len(numbers)

def standard_deviation(numbers):
    avg = average(numbers)
    variance = 0
    for num in numbers:
        variance += (num - avg) ** 2
    variance = variance / len(numbers)
    return variance ** 0.5 

data = [10, 12, 23, 23, 16, 23, 21, 16]
print(f"標準差：{standard_deviation(data):.2f}")

    if "@" not in email:
        return False
    if "." not in email:
        return False
    at_index = email.index("@")
    if at_index == 0 or at_index == len(email) - 1:
        return False
    return True

def normalize(numbers):
    if len(numbers) == 0:
        return []
    
    max_val = max(numbers)
    min_val = min(numbers)
    
    if max_val == min_val:
        return [0] * len(numbers)
    
    normalized = []
    for num in numbers:
        norm = (num - min_val) / (max_val - min_val)
        normalized.append(norm)
    return normalized

    n = len(sorted_list)
    
    for i in range(n):
        min_idx = i

def binary_search(sorted_list, target):
    left = 0
    right = len(sorted_list) - 1
    
    while left <= right:
        mid = (left + right) // 2
        
        if sorted_list[mid] == target:
            return mid
        elif sorted_list[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return -1

def bubble_sort(numbers):
    sorted_list = numbers.copy()
    n = len(sorted_list)
    for i in range(n):
        for j in range(0, n - i - 1):
            if sorted_list[j] > sorted_list[j + 1]:
                temp = sorted_list[j]
                sorted_list[j] = sorted_list[j + 1]
                sorted_list[j + 1] = temp
    return sorted_list

def find_median(numbers):
    sorted_nums = bubble_sort(numbers)
    n = len(sorted_nums)
    
    if n % 2 == 0:
        mid1 = sorted_nums[n // 2 - 1]
        mid2 = sorted_nums[n // 2]
        return (mid1 + mid2) / 2
    else:
        return sorted_nums[n // 2]

data = [3, 1, 4, 1, 5, 9, 2, 6]
print(f"中位數：{find_median(data)}")
