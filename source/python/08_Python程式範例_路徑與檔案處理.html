with open("products.csv", "r", encoding="utf-8") as file:
    lines = file.readlines()
    errors = []
    for i, line in enumerate(lines[1:], start=2):
        values = line.strip().split(",")
        if len(values) != 3:
            errors.append(f"第{i}行：欄位數量錯誤")
        else:
            try:
                price = int(values[1])
                stock = int(values[2])
                if price < 0:
                    errors.append(f"第{i}行：價格不可為負")
                if stock < 0:
                    errors.append(f"第{i}行：庫存不可為負")
            except:
                errors.append(f"第{i}行：數值格式錯誤")

if len(errors) == 0:
    print("資料驗證通過")
else:
    print("發現錯誤：")
    for error in errors:
        print(f"  {error}")

host=localhost

with open("settings.ini", "r", encoding="utf-8") as file:
    current_section = None
    settings = {}
    
    for line in file:
        line = line.strip()
        if line.startswith("[") and line.endswith("]"):
            current_section = line[1:-1]
            settings[current_section] = {}
        elif "=" in line and current_section:
            key, value = line.split("=")
            settings[current_section][key] = value

print(f"INI 設定：{settings}")

def update_stock(filename, product_name, quantity_change):
    products = []
    
    with open(filename, "r", encoding="utf-8") as file:
        lines = file.readlines()
        header = lines[0]
        for line in lines[1:]:
            values = line.strip().split(",")
            products.append({
                "name": values[0],
                "price": int(values[1]),
                "stock": int(values[2])
            })
    
    for product in products:
        if product["name"] == product_name:
            product["stock"] += quantity_change
    
    with open(filename, "w", encoding="utf-8") as file:
        file.write(header)
        for product in products:
            line = f"{product['name']},{product['price']},{product['stock']}\n"
            file.write(line)

def get_balance(filename):
    balance = 0
    try:
        with open(filename, "r", encoding="utf-8") as file:
            for line in file:
                parts = line.strip().split(",")
                trans_type = parts[1]
                amount = float(parts[2])
                if trans_type == "存款":
                    balance += amount
                elif trans_type == "提款":
                    balance -= amount
    except:
        pass
    return balance

def import_with_validation(filename):
    valid_data = []
    errors = []
    
    with open(filename, "r", encoding="utf-8") as file:
        lines = file.readlines()
        for i, line in enumerate(lines, 1):
            parts = line.strip().split(",")
            
            if len(parts) != 3:
                errors.append(f"第{i}行：欄位數量錯誤")
                continue
            
            try:
                score = int(parts[2])
                if score < 0 or score > 100:
                    errors.append(f"第{i}行：分數超出範圍")
                    continue
                
                valid_data.append({
                    "id": parts[0],
                    "name": parts[1],
                    "score": score
                })
            except:
                errors.append(f"第{i}行：分數格式錯誤")
    
    return valid_data, errors

valid, errors = import_with_validation("grades.txt")
print(f"有效資料：{len(valid)} 筆")

def incremental_backup(filename, backup_dir):
    backup_count = 1
    backup_name = f"{backup_dir}/backup_{backup_count}.txt"
    
    while True:
        try:
            with open(backup_name, "r", encoding="utf-8"):
                backup_count += 1
                backup_name = f"{backup_dir}/backup_{backup_count}.txt"
        except:
            break
    
    with open(filename, "r", encoding="utf-8") as source:
        content = source.read()
    
    with open(backup_name, "w", encoding="utf-8") as backup:
        backup.write(content)
    
    print(f"備份到：{backup_name}")

def search_in_files(keyword, filenames):
    results = []
    
    for filename in filenames:
        try:
            with open(filename, "r", encoding="utf-8") as file:
                line_num = 1
                for line in file:
                    if keyword in line:
                        results.append({
                            "file": filename,
                            "line": line_num,
                            "content": line.strip()
                        })

        except:
            pass
    
    return results

def process_pipeline(input_file, output_file):
    with open(input_file, "r", encoding="utf-8") as file:
        lines = file.readlines()
    
    processed = []
    for line in lines[1:]:
        values = line.strip().split(",")
        name = values[0]

        grade = "A" if score >= 90 else "B" if score >= 80 else "C"
        
        processed.append(f"{name},{score},{grade}\n")
    
    with open(output_file, "w", encoding="utf-8") as file:
        file.write("姓名,分數,等級\n")

def batch_process_files(file_list, operation):
    results = []
    
    for filename in file_list:
        try:
            with open(filename, "r", encoding="utf-8") as file:
                content = file.read()
                result = operation(content)
                results.append({"file": filename, "result": result})

    return results

def calculate_checksum(filename):
    checksum = 0
    with open(filename, "r", encoding="utf-8") as file:
        content = file.read()
        for char in content:

def remove_duplicates(input_file, output_file):
    with open(input_file, "r", encoding="utf-8") as file:
        lines = file.readlines()
    
    unique_lines = []
    seen = []
    
    for line in lines:
        if line not in seen:
            unique_lines.append(line)
            seen.append(line)
    
    with open(output_file, "w", encoding="utf-8") as file:
        for line in unique_lines:
            file.write(line)
    
    print(f"原始：{len(lines)} 行，去重後：{len(unique_lines)} 行")

class FileManager:
    def __init__(self, base_dir):
        self.base_dir = base_dir
    
    def create_file(self, filename, content):
        path = f"{self.base_dir}/{filename}"
        with open(path, "w", encoding="utf-8") as file:
            file.write(content)
        return f"已建立：{path}"
    
    def read_file(self, filename):
        path = f"{self.base_dir}/{filename}"
        try:
            with open(path, "r", encoding="utf-8") as file:
                return file.read()
        except:
            return "檔案不存在"
    
    def list_files(self):
        return ["使用 os 模組可實現完整功能"]
    
    def delete_file(self, filename):
        return f"已刪除：{filename}"
