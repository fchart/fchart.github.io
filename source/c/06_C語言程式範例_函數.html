void to_binary(int number, char *binary) {
    if (number == 0) {
        binary[0] = '0';
        binary[1] = '\0';
        return;
    }
    char temp[32];
    int idx = 0;
    while (number > 0) {
        temp[idx++] = (number % 2) + '0';
        number /= 2;
    }
    for (int i = 0; i < idx; i++) {
        binary[i] = temp[idx - 1 - i];
    }
    binary[idx] = '\0';
}

題目 37：排序函數（氣泡排序）
#include <stdio.h>

void bubble_sort(int numbers[], int size, int sorted[]) {
    // 使用 for 迴圈逐一複製元素
    for (int i = 0; i < size; i++) {
        sorted[i] = numbers[i];
    }    
    // 氣泡排序巢狀迴圈
    for (int i = 0; i < size; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                int temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }
}

void print_array(int arr[], int size) {
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);
    printf("\n");
}

int main() {
    printf("\n題目37：氣泡排序\n");
    int data[] = {64, 34, 25, 12, 22, 11, 90};
    int size = sizeof(data) / sizeof(data[0]);
    int sorted[7];
    bubble_sort(data, size, sorted);
    printf("排序前：");
    print_array(data, size);
    printf("排序後：");
    print_array(sorted, size);
    return 0;
}

題目 39：標準差計算函數
#include <stdio.h>
#include <math.h>

double average(int numbers[], int size) {
    int total = 0;
    for (int i = 0; i < size; i++) total += numbers[i];
    return (double)total / size;
}

double standard_deviation(int numbers[], int size) {
    double avg = average(numbers, size);
    double variance = 0;
    for (int i = 0; i < size; i++) {
        variance += (numbers[i] - avg) * (numbers[i] - avg);
    }
    variance /= size;

    double max_val = numbers[0], min_val = numbers[0];

        if (numbers[i] > max_val) max_val = numbers[i];
        if (numbers[i] < min_val) min_val = numbers[i];

        for (int i = 0; i < size; i++) result[i] = 0;
        return;

    }
}

int binary_search(int sorted_list[], int size, int target) {
    int left = 0, right = size - 1;
    while (left <= right) {
        int mid = (left + right) / 2;
        if (sorted_list[mid] == target) return mid;
        else if (sorted_list[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

int word_frequency(const char *text, char words[][20], int counts[]) {
    char buffer[200];
    // 手動複製 text 到 buffer
    int idx = 0;
    while (text[idx] != '\0' && idx < sizeof(buffer) - 1) {
        buffer[idx] = tolower((unsigned char)text[idx]);
        idx++;
    }
    buffer[idx] = '\0';
    int freq_count = 0;
    int start = 0;
    for (int i = 0; ; i++) {
        if (buffer[i] == ' ' || buffer[i] == '\0') {
            if (i > start) {
                char token[20];
                int len = i - start;
                if (len >= sizeof(token)) len = sizeof(token) - 1; // 避免溢位
                for (int k = 0; k < len; k++) token[k] = buffer[start + k];
                token[len] = '\0';
                int found = 0;
                for (int j = 0; j < freq_count; j++) {
                    if (strncmp(words[j], token, strlen(words[j])) == 0) {
                        counts[j]++;
                        found = 1;
                        break;
                    }
                }
                if (!found) {
                    int copy_len = strlen(token);
                    if (copy_len >= 20) copy_len = 19;
                    for (int k = 0; k < copy_len; k++) {
                        words[freq_count][k] = token[k];
                    }
                    words[freq_count][copy_len] = '\0';
                    counts[freq_count] = 1;
                    freq_count++;
                }
            }
            start = i + 1;
        }
        if (buffer[i] == '\0') break;
    }
    return freq_count;
}

void bubble_sort(int numbers[], int size, int sorted[]) {
    for (int i = 0; i < size; i++) {
        sorted[i] = numbers[i];
    }
    for (int i = 0; i < size; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                int temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }
}

double find_median(int numbers[], int size) {
    int sorted_nums[100];
    bubble_sort(numbers, size, sorted_nums);
    if (size % 2 == 0) {
        int mid1 = sorted_nums[size / 2 - 1];
        int mid2 = sorted_nums[size / 2];
        return (mid1 + mid2) / 2.0;
    } else {
        return sorted_nums[size / 2];
    }
}
