#include <stdio.h>

struct Student {
    char name[20];
    int score;
};

int main() {
    struct Student students[3] = {
        {"Amy", 88}, {"Bob", 76}, {"Cathy", 95}
    };
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2 - i; j++) {
            if (students[j].score > students[j + 1].score) {
                struct Student temp = students[j];
                students[j] = students[j + 1];
                students[j + 1] = temp;
            }
        }
    }
    printf("題目26：排序後最低分為 %s（%d分）\n", students[0].name, students[0].score);
    return 0;
}

#include <math.h>

struct Point {
    double x;
    double y;
};

int main() {
    struct Point poly[4] = {{0, 0}, {4, 0}, {4, 3}, {0, 3}};
    double perimeter = 0;
    for (int i = 0; i < 4; i++) {
        struct Point p1 = poly[i];
        struct Point p2 = poly[(i + 1) % 4];
        perimeter += sqrt(pow(p2.x - p1.x, 2) + pow(p2.y - p1.y, 2));
    }
    printf("題目36：多邊形周長 = %g\n", perimeter);
    return 0;
}
