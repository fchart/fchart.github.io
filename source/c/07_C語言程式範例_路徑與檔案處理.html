    char line[256];
    int price;
} Row;

FILE *file = fopen("products.csv", "r");
if (file == NULL) return 1;
char header[256];

Row rows[100];
int count = 0;

    sscanf(rows[count].line, "%31[^,],%d,%d", name, &rows[count].price, &stock);
    count++;

fclose(file);

/* 依價格由高到低排序（氣泡排序） */
for (int i = 0; i < count - 1; i++) {
    for (int j = 0; j < count - 1 - i; j++) {
        if (rows[j].price < rows[j + 1].price) {
            Row temp = rows[j];
            rows[j] = rows[j + 1];
            rows[j + 1] = temp;
        }
    }
}

FILE *out = fopen("products_sorted.csv", "w");
fprintf(out, "%s", header);
for (int i = 0; i < count; i++) fprintf(out, "%s", rows[i].line);
fclose(out);
