題目 6：尋找最大值與最小值
#include <stdio.h>

int main() {
    printf("\n題目6：最大值與最小值\n");
    int numbers[] = {42, 17, 89, 5, 33, 71};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    int max = numbers[0], min = numbers[0];

    for (int i = 1; i < size; i++) {
        if (numbers[i] > max) max = numbers[i];  // 更新最大值
        if (numbers[i] < min) min = numbers[i];  // 更新最小值
    }

    printf("陣列：");
    for (int i = 0; i < size; i++) printf("%d ", numbers[i]);
    printf("\n最大值：%d，最小值：%d\n", max, min);
    return 0;
}

題目 10：真正反轉陣列（交換）
#include <stdio.h>

int main() {
    printf("\n題目10：原地反轉陣列\n");
    int numbers[] = {1, 2, 3, 4, 5};
    int size = sizeof(numbers) / sizeof(numbers[0]);

    for (int i = 0; i < size / 2; i++) {
        int tmp = numbers[i];                    // 交換前後對稱元素
        numbers[i] = numbers[size - 1 - i];
        numbers[size - 1 - i] = tmp;
    }

    printf("反轉後：");
    for (int i = 0; i < size; i++) printf("%d ", numbers[i]);
    printf("\n");
    return 0;
}

    int size = sizeof(numbers) / sizeof(numbers[0]);
    int target = 20;
    int index = -1;

    for (int i = 0; i < size; i++) {
        if (numbers[i] == target) { index = i; break; }  // 找到即記錄索引並跳出

}

題目 38：傳回指標的函式
#include <stdio.h>

int *find_max(int arr[], int n) {
    int *m = &arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > *m) m = &arr[i];  // 記錄最大元素的位址
    }
    return m;  // 傳回指標
}

int main() {
    printf("\n題目38：傳回指標\n");
    int numbers[] = {3, 7, 2, 9, 5};
    int *max = find_max(numbers, 5);
    printf("最大值：%d\n", *max);
    return 0;
}

題目 51：二分搜尋
#include <stdio.h>

int binary_search(int arr[], int n, int target) {
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = (low + high) / 2;
        if (arr[mid] == target) return mid;      // 找到
        else if (arr[mid] < target) low = mid + 1;  // 往右半邊
        else high = mid - 1;                        // 往左半邊
    }
    return -1;
}

int main() {
    printf("\n題目51：二分搜尋\n");
    int sorted[] = {2, 4, 6, 8, 10, 12, 14};
    int index = binary_search(sorted, 7, 10);
    printf("找到 10 位於索引：%d\n", index);
    return 0;
}

void minmax(int *arr, int n, int *min, int *max) {

    *max = arr[0];

void reverse(int *arr, int n) {
    int left = 0;               // 起始索引
    int right = n - 1;          // 末端索引
    while (left < right) {      // 兩端向中間靠攏
        int tmp = arr[left];
        arr[left] = arr[right];
        arr[right] = tmp;
        left++;
        right--;
    }
}

題目 78：字串反轉並比較
#include <stdio.h>
#include <string.h>

void reverse_str(char *s) {
    int n = strlen(s);
    for (int i = 0; i < n / 2; i++) {
        char t = s[i];
        s[i] = s[n - 1 - i];
        s[n - 1 - i] = t;
    }
}

int strings_equal(const char *a, const char *b) {
    int i = 0;
    while (a[i] != '\0' && b[i] != '\0') {
        if (a[i] != b[i]) return 0;  // 某字元不同
        i++;
    }
    return (a[i] == '\0' && b[i] == '\0') ? 1 : 0;  // 都要到結尾
}

int main() {
    printf("\n題目78：比較字串與其反轉\n");
    char original[] = "level";
    char reversed[10];

    // 手刻複製取代 strcpy()，避免 JSCPP 複製後無法再改單一字元
    int k = 0;
    while (original[k] != '\0') { reversed[k] = original[k]; k++; }
    reversed[k] = '\0';

    reverse_str(reversed);

    printf("原文：%s\n反轉：%s\n", original, reversed);
    printf("是否回文：%s\n", strings_equal(original, reversed) ? "是" : "否");
    return 0;
}

題目 80：綜合 - 學生資料分析
#include <stdio.h>
#include <string.h>

int main() {
    printf("\n題目80：學生資料分析\n");
    char names[][10] = {"Alice", "Bob", "Charlie", "David"};
    int scores[] = {85, 92, 70, 60};
    int n = 4;

    // 計算平均
    int total = 0;
    for (int i = 0; i < n; i++) total += scores[i];
    double avg = total / (double)n;

    // 找出最高分
    int max = scores[0];
    char top_name[10];
    strcpy(top_name, names[0]);
    for (int i = 1; i < n; i++) {
        if (scores[i] > max) { max = scores[i]; strcpy(top_name, names[i]); }
    }

    // 統計及格與不及格
    int passed = 0;
    for (int i = 0; i < n; i++) if (scores[i] >= 60) passed++;

    printf("全班平均：%.2f\n", avg);
    printf("最高分：%s（%d 分）\n", top_name, max);
    printf("及格人數：%d / %d\n", passed, n);
    return 0;
}
