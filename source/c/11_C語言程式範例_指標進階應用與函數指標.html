題目 16：使用指標實作陣列反轉
#include <stdio.h>

void reverseArray(int *arr, int n) {
    int *start = arr;
    int *end = arr + n - 1;
    while (start < end) {
        int temp = *start;
        *start = *end;
        *end = temp;
        start++;
        end--;
    }
}

int main() {
    int nums[5] = {1, 2, 3, 4, 5};
    reverseArray(nums, 5);
    printf("題目16：反轉後第一個元素 = %d\n", nums[0]);
    return 0;
}

題目 35：使用函數指標實作策略模式（排序策略切換）
#include <stdio.h>

int ascending(int a, int b) { return a - b; }

void bubbleSort(int arr[], int n, int (*compare)(int, int)) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (compare(arr[j], arr[j + 1]) > 0) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

int main() {
    int arr[5] = {5, 2, 8, 1, 9};
    bubbleSort(arr, 5, ascending);
    printf("題目35：升冪排序後第一個元素 = %d\n", arr[0]);
    return 0;
}

char* myStrchr(const char *str, char target) {

        if (*str == target) return (char *)str;
        str++;
    }
    return NULL;
}

void reverseString(char *str) {
    char *start = str;
    char *end = str + strlen(str) - 1;
    while (start < end) {
        char temp = *start;
        *start = *end;
        *end = temp;
        start++;
        end--;
    }
}

int main() {

題目 53：字串指標陣列的排序後搜尋（二分搜尋法）
#include <stdio.h>
#include <string.h>

int main() {
    const char *names[5] = {"Amy", "Bob", "Cathy", "David", "Emma"};  // 已排序
    const char *target = "Cathy";
    int low = 0, high = 4, found = -1;
    while (low <= high) {
        int mid = (low + high) / 2;
        int cmp = strcmp(names[mid], target);
        if (cmp == 0) { found = mid; break; }
        else if (cmp < 0) low = mid + 1;
        else high = mid - 1;
    }
    printf("題目53：二分搜尋『%s』找到索引 = %d\n", target, found);
    return 0;
}

int isPalindrome(const char *str) {
    const char *start = str;
    const char *end = str + strlen(str) - 1;
    while (start < end) {
        if (*start != *end) return 0;
        start++;
        end--;
    }
    return 1;
}
