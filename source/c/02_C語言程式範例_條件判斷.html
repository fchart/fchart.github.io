    int limit = 60;
    int fine;
    printf("題目57 - 請輸入行駛速度（公里/小時）：");
    scanf("%d", &speed);
    if (speed > limit + 40) {
        fine = 3600;
        printf("超速 %d 公里，罰款 %d 元\n", speed - limit, fine);
