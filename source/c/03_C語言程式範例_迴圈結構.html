題目 49：密碼驗證（限制次數）

#include <string.h>

int main() {
    printf("\n題目49：密碼驗證（最多 3 次）\n");
    char password[20] = "c12345";
    char user_input[20];
    int attempts = 0;
    int max_attempts = 3;
    int success = 0;
    while (attempts < max_attempts) {
        printf("請輸入密碼：");
        scanf("%s", user_input);
        attempts++;
        if (strncmp(user_input, password, strlen(password)) == 0) {
            printf("登入成功\n");
            success = 1;
            break;
        } else {
            printf("密碼錯誤，剩餘 %d 次機會\n", max_attempts - attempts);
        }
    }
    if (!success) {
        printf("登入失敗，帳號已鎖定\n");
    }
    return 0;
}

題目 70：計算標準差
#include <stdio.h>
#include <math.h>

int main() {
    printf("\n題目70：計算標準差\n");
    int n;
    printf("請輸入數據個數：");
    scanf("%d", &n);
    float numbers[n];
    float total = 0;
    for (int i = 0; i < n; i++) {
        printf("請輸入第 %d 個數據：", i + 1);
        float temp;
        scanf("%f", &temp);  // 不能直接使用陣列元素
        numbers[i] = temp;
        total += numbers[i];
    }
    float mean = total / n;
    float variance = 0;
    for (int i = 0; i < n; i++) {
        variance += (numbers[i] - mean) * (numbers[i] - mean);
    }
    variance = variance / n;
    float std_dev = sqrt(variance);
    printf("平均數 = %.2f，標準差 = %.2f\n", mean, std_dev);
    return 0;
}

    printf("\n題目75：印出帕斯卡三角形\n");
    int n;
    printf("請輸入層數：");
    scanf("%d", &n);
    int row[n];  // C99 VLA，儲存上一列數值
    for (int i = 0; i < n; i++) {
        int new_row[n];
        new_row[0] = 1;
        for (int j = 1; j < i; j++) {
            new_row[j] = row[j - 1] + row[j];
        }
        if (i > 0) {
            new_row[i] = 1;
        }
        for (int j = 0; j < n - i; j++) printf(" ");
        for (int j = 0; j <= i; j++) printf("%d ", new_row[j]);
        printf("\n");
        for (int j = 0; j <= i; j++) row[j] = new_row[j];
    }
    return 0;
}

        printf("請選擇功能：");
        scanf("%d", &choice);
        if (choice == 1) {
            int amount;
            printf("請輸入存款金額：");
            scanf("%d", &amount);
            balance += amount;
            printf("存款成功，餘額 = %d 元\n", balance);
        } else if (choice == 2) {
            int amount;
            printf("請輸入提款金額：");
            scanf("%d", &amount);
            if (amount > balance) {
                printf("餘額不足\n");
            } else {
                balance -= amount;
                printf("提款成功，餘額 = %d 元\n", balance);
            }
        } else if (choice == 3) {
            printf("目前餘額 = %d 元\n", balance);
        } else if (choice == 4) {
            printf("謝謝使用，再見！\n");
            break;
        } else {
            printf("無效選擇\n");
        }
    }

}
