題目 22：動態陣列 push_back 新增元素
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
    int capacity;
} DynArray;

void push_back(DynArray *arr, int value) {
    arr->data[arr->size] = value;
    arr->size++;
}

int main() {
    DynArray arr;
    arr.capacity = 5;
    arr.size = 0;
    arr.data = (int *)malloc(arr.capacity * sizeof(int));
    push_back(&arr, 10);

}
題目 23：動態陣列容量不足時 realloc 擴充（倍增策略）
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
    int capacity;
} DynArray;

void push_back(DynArray *arr, int value) {
    if (arr->size == arr->capacity) {
        arr->capacity *= 2;
        arr->data = (int *)realloc(arr->data, arr->capacity * sizeof(int));
    }
    arr->data[arr->size] = value;
    arr->size++;
}

int main() {
    DynArray arr;
    arr.capacity = 2;
    arr.size = 0;
    arr.data = (int *)malloc(arr.capacity * sizeof(int));
    for (int i = 0; i < 5; i++) push_back(&arr, i + 1);
    printf("題目23：擴充後容量 = %d，元素數量 = %d\n", arr.capacity, arr.size);
    free(arr.data);
    return 0;
}
題目 24：動態陣列刪除最後一個元素（pop_back）
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *data;
    int size;
    int capacity;
} DynArray;

int pop_back(DynArray *arr) {
    arr->size--;
    return arr->data[arr->size];
}

int main() {
    int init[] = {5, 10, 15};
    DynArray arr;
    arr.capacity = 3;
    arr.size = 3;
    arr.data = (int *)malloc(arr.capacity * sizeof(int));
    for (int i = 0; i < 3; i++) arr.data[i] = init[i];
    int popped = pop_back(&arr);
    printf("題目24：彈出的元素 = %d，剩餘大小 = %d\n", popped, arr.size);
    free(arr.data);
    return 0;
}

    int original[] = {29, 10, 45, 3, 18};
    for (int i = 0; i < n; i++) arr[i] = original[i];
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;
        for (int j = i + 1; j < n; j++) {

}

    int *arr = (int *)malloc(capacity * sizeof(int));
    int size = 0, reallocCount = 0;
    for (int i = 0; i < 20; i++) {
        if (size == capacity) {
            capacity *= 2;
            arr = (int *)realloc(arr, capacity * sizeof(int));
            reallocCount++;
        }
        arr[size++] = i;

題目 80：綜合應用：利用二元樹（BST）中序走訪得到排序結果
#include <stdio.h>
#include <stdlib.h>

struct TreeNode {
    int data;
    struct TreeNode *left;
    struct TreeNode *right;
};

struct TreeNode* insert(struct TreeNode *node, int value) {
    if (node == NULL) {
        struct TreeNode *newNode = (struct TreeNode *)malloc(sizeof(struct TreeNode));
        newNode->data = value;
        newNode->left = newNode->right = NULL;
        return newNode;
    }
    if (value < node->data) node->left = insert(node->left, value);
    else node->right = insert(node->right, value);
    return node;
}

int sortedArr[10];
int idx = 0;

void inorderFill(struct TreeNode *node) {
    if (node == NULL) return;
    inorderFill(node->left);
    sortedArr[idx++] = node->data;
    inorderFill(node->right);
}

int main() {
    struct TreeNode *root = NULL;
    int values[] = {40, 10, 60, 5, 25, 50, 70};
    for (int i = 0; i < 7; i++) root = insert(root, values[i]);
    inorderFill(root);
    printf("題目80：利用BST排序後的第一個與最後一個元素 = %d, %d\n", sortedArr[0], sortedArr[idx - 1]);
    return 0;
}
