    if (n < 2) {
        is_prime = 0;
    }

題目 76：回文數判斷
#include <stdio.h>

int main() {
    int n = 121;
    int reverse = 0;
    int temp = n;
    while (temp > 0) {
        reverse = reverse * 10 + temp % 10;
        temp = temp / 10;
    }
    printf("%d 是回文數: %d\n", n, reverse == n);
    return 0;
}
